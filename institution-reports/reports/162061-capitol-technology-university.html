standing = query("""
    WITH peer_group AS (
        SELECT m.*, m.unitid = $unitid AS is_focus
        FROM metrics AS m
        WHERE m.metric IN ('grad_rate', 'retention_rate')
          AND m.year = (SELECT max(year) FROM metrics WHERE metric = 'grad_rate')
          AND (m.unitid = $unitid
               OR m.unitid IN (SELECT peer_unitid FROM peer_pairs WHERE unitid = $unitid))
    )
    SELECT
        g.metric,
        i.inst_name,
        i.state_abbr,
        g.is_focus,
        g.value,
        g.base,
        -- share of the group with a lower value
        percent_rank() OVER (PARTITION BY g.metric ORDER BY g.value) AS pct_rank,
        count(*) OVER (PARTITION BY g.metric) - 1 AS n_peers,
        median(g.value) FILTER (NOT g.is_focus) OVER (PARTITION BY g.metric) AS peer_median
    FROM peer_group AS g
    JOIN institutions AS i USING (unitid)
    ORDER BY g.metric, g.value
""")

enrollment = query("""
    WITH peer_group AS (
        SELECT $unitid AS unitid
        UNION ALL
        SELECT peer_unitid FROM peer_pairs WHERE unitid = $unitid
    ),
    indexed AS (
        SELECT
            m.unitid,
            m.year,
            m.value AS headcount,
            m.value / first_value(m.value) OVER by_inst - 1 AS change,
            min(m.year) OVER by_inst AS first_year
        FROM metrics AS m
        JOIN peer_group USING (unitid)
        WHERE m.metric = 'enrollment'
        WINDOW by_inst AS (PARTITION BY m.unitid ORDER BY m.year)
    )
    -- keep only institutions observed from the first year, so every index
    -- starts from the same base
    SELECT
        year,
        max(headcount) FILTER (unitid = $unitid) AS headcount,
        max(change) FILTER (unitid = $unitid) AS value,
        quantile_cont(change, 0.25) FILTER (unitid <> $unitid) AS peer_p25,
        median(change) FILTER (unitid <> $unitid) AS peer_median,
        quantile_cont(change, 0.75) FILTER (unitid <> $unitid) AS peer_p75,
        count(*) FILTER (unitid <> $unitid) AS n_peers
    FROM indexed
    WHERE first_year = (SELECT min(year) FROM metrics WHERE metric = 'enrollment')
    GROUP BY year
    ORDER BY year
""")

focus = standing[standing.is_focus].set_index("metric")


def tile(label: str, value: str, lines: list[str]) -> str:
    detail = "".join(f"<div class='tile-detail'>{line}</div>" for line in lines)
    return f"<div class='tile'><div class='tile-label'>{label}</div><div class='tile-value'>{value}</div>{detail}</div>"


def rate_tile(metric: str, label: str) -> str:
    if metric not in focus.index:
        return tile(label, "n/a", ["Not reported, or cohort under 30"])

trends = query("""
    WITH peer AS (
        SELECT
            m.metric,
            m.year,
            quantile_cont(m.value, 0.25) AS peer_p25,
            median(m.value) AS peer_median,
            quantile_cont(m.value, 0.75) AS peer_p75,

        JOIN metrics AS m ON m.unitid = p.peer_unitid

def trend_chart(df: pd.DataFrame, y_title: str, fmt: str, domain=None, x_title="Year") -> alt.LayerChart:
    """Institution line over the peer median and the peers' interquartile band."""
    x = alt.X("year:O", title=x_title, axis=alt.Axis(labelAngle=0, grid=False))
    y_scale = alt.Scale(domain=domain) if domain else alt.Scale(zero=True, nice=True)
    base = alt.Chart(df)

    band = base.mark_area(color=PEER, opacity=0.16).encode(
        x=x, y=alt.Y("peer_p25:Q", title=y_title, scale=y_scale, axis=alt.Axis(format=fmt)),
        y2="peer_p75:Q",
    )
    lines = (
        base.transform_fold(["value", "peer_median"], as_=["series", "y"])
        .transform_calculate(series=f"datum.series == 'value' ? {NAME!r} : {PEER_LABEL!r}")
        .mark_line(strokeWidth=2.5, point=alt.OverlayMarkDef(size=45, filled=True, stroke="white", strokeWidth=1.5))
        .encode(
            x=x, y="y:Q",
            color=alt.Color("series:N", scale=alt.Scale(domain=[NAME, PEER_LABEL], range=[FOCUS, PEER])),
        )
    )
    hover = alt.selection_point(on="pointerover", nearest=True, fields=["year"], empty=False, clear="pointerout")
    crosshair = (
        base.mark_rule(color=MUTED, strokeWidth=1)
        .encode(
            x=x,
            opacity=alt.condition(hover, alt.value(0.6), alt.value(0)),
            tooltip=[
                alt.Tooltip("year:O", title="Year"),
                alt.Tooltip("value:Q", title=NAME, format=fmt),
                alt.Tooltip("peer_median:Q", title=PEER_LABEL, format=fmt),
                alt.Tooltip("peer_p25:Q", title="Peers, 25th percentile", format=fmt),
                alt.Tooltip("peer_p75:Q", title="Peers, 75th percentile", format=fmt),
                alt.Tooltip("n_peers:Q", title="Peers reporting"),
            ],
        )
        .add_params(hover)
    )
    return (band + lines + crosshair).properties(width=WIDTH, height=280)


def trend_sentence(df: pd.DataFrame, noun: str) -> str:
    own = df.dropna(subset=["value"])
    if len(own) < 2:
        return f"{NAME} does not have enough reported years to show a trend in its {noun}."
    a, b = own.iloc[0], own.iloc[-1]
    return (
        f"{NAME}'s {noun} {moved(a.value, b.value)} between {int(a.year)} and {int(b.year)}, "
        f"from {pct(a.value)} to {pct(b.value)}. Over the same years the peer median "
        f"{moved(a.peer_median, b.peer_median)}, leaving {NAME} {gap(b.value, b.peer_median)} "
        f"the peer median of {pct(b.peer_median)}."
    )

    WITH peer AS (
        SELECT
            g.race,
            sum(g.completers) / sum(g.cohort) AS peer_rate,
            sum(g.cohort) AS peer_cohort
        FROM peer_pairs AS p
        JOIN grad_by_race AS g ON g.unitid = p.peer_unitid
        WHERE p.unitid = $unitid
        GROUP BY g.race
    )
    SELECT
        g.race,
        g.sort_order,
        g.cohort,
        g.completers / g.cohort AS rate,
        peer.peer_rate,
        peer.peer_cohort,
        g.cohort >= min_cohort() AS reportable
    FROM grad_by_race AS g
    JOIN peer USING (race)
    WHERE g.unitid = $unitid
    ORDER BY g.sort_order
""")

Data and methods

Source. Integrated Postsecondary Education Data System (IPEDS), National Center for Education Statistics, retrieved through the Urban Institute Education Data Portal. Years 2013 to 2023. The snapshot lives in data/ as Parquet and is queried with DuckDB; scripts/fetch_ipeds.py rebuilds it.

Measures.

  • Six-year graduation rate: share of full-time, first-time, bachelor’s-seeking students who completed at the same institution within 150% of normal time. The year is the reporting year; the cohort entered six years earlier.
  • First-year retention rate: share of full-time, first-time students from the previous fall who returned the next fall.
  • Fall enrollment: total headcount, all levels and attendance statuses.

Peers. Every other institution with the same control (private nonprofit) and 2021 Carnegie Basic classification (Doctoral/Professional), nationwide. Peer statistics for each year use the peers that reported that year, so the set shifts slightly over time.

Small cohorts. Rates based on fewer than 30 students are left out, for this institution and for its peers.

Limits. These rates follow first-time, full-time students only; transfer-in and part-time students are not counted, and students who transfer out and finish elsewhere count as non-completers. The peer group is a classification match, not a statistical one: it does not adjust for selectivity or student background.

```{python}
def pct(x: float) -> str:
    return f"{x:.0%}"


def signed(x: float) -> str:
    return f"{x:+.0%}".replace("-", "\u2212")


def gap(a: float, b: float, unit: str = "points") -> str:
    """'4 points above' / 'level with', for a pair of rates."""
    diff = round((a - b) * 100)
    if diff == 0:
        return "level with"
    return f"{abs(diff)} {unit[:-1] if abs(diff) == 1 else unit} {'above' if diff > 0 else 'below'}"


def moved(a: float, b: float) -> str:
    """'rose 4 points' / 'was unchanged', from a to b."""
    diff = round((b - a) * 100)
    if diff == 0:
        return "was unchanged"
    return f"{'rose' if diff > 0 else 'fell'} {abs(diff)} point{'' if abs(diff) == 1 else 's'}"
```

```{python}
    own = df.dropna(subset=["value"])
    if len(own) < 2:
        return f"{NAME} does not have enough reported years to show a trend in its {noun}."
    a, b = own.iloc[0], own.iloc[-1]
    return (
        f"{NAME}'s {noun} {moved(a.value, b.value)} between {int(a.year)} and {int(b.year)}, "
        f"from {pct(a.value)} to {pct(b.value)}. Over the same years the peer median "
        f"{moved(a.peer_median, b.peer_median)}, leaving {NAME} {gap(b.value, b.peer_median)} "
        f"the peer median of {pct(b.peer_median)}."
    )


BAND_NOTE = "The shaded band covers the middle half of peer institutions (25th to 75th percentile)."
```

```{python}

display(Markdown(trend_sentence(grad, "six-year graduation rate") + " " + BAND_NOTE))
trend_chart(grad, "Six-year graduation rate", ".0%", domain=[0, 1], x_title="Reporting year (cohort entered six years earlier)")
```

## Data and methods

```{python}
#| label: methods
years = query("SELECT min(year) AS first, max(year) AS last FROM metrics WHERE unitid = $unitid").iloc[0]
Markdown(f"""
**Source.** Integrated Postsecondary Education Data System (IPEDS), National Center for
Education Statistics, retrieved through the Urban Institute
[Education Data Portal](https://educationdata.urban.org/documentation/colleges.html).
Years {int(years['first'])} to {int(years['last'])}. The snapshot lives in `data/` as Parquet and is
queried with DuckDB; `scripts/fetch_ipeds.py` rebuilds it.

**Measures.**

- *Six-year graduation rate*: share of full-time, first-time, bachelor's-seeking students who completed
  at the same institution within 150% of normal time. The year is the reporting year; the cohort
  entered six years earlier.
- *First-year retention rate*: share of full-time, first-time students from the previous fall who
  returned the next fall.
- *Fall enrollment*: total headcount, all levels and attendance statuses.

**Peers.** Every other institution with the same control ({inst.control_label.lower()}) and 2021
Carnegie Basic classification ({inst.carnegie_label}), nationwide. Peer statistics for each year use
the peers that reported that year, so the set shifts slightly over time.

**Small cohorts.** Rates based on fewer than 30 students are left out, for this institution and
for its peers.

**Limits.** These rates follow first-time, full-time students only; transfer-in and part-time students
are not counted, and students who transfer out and finish elsewhere count as non-completers. The
peer group is a classification match, not a statistical one: it does not adjust for selectivity or
student background.
""")
```
