summary = con.execute("""
    WITH latest AS (
        SELECT unitid, value, base
        FROM metrics
        WHERE metric = 'grad_rate'
          AND year = (SELECT max(year) FROM metrics WHERE metric = 'grad_rate')
    )
    SELECT
        u.inst_name,
        u.city,
        u.control_label,
        u.carnegie_label,
        u.report_file,
        own.value AS grad_rate,
        own.base AS cohort,
        median(peer.value) AS peer_median,
        count(*) AS n_peers,
        avg((peer.value < own.value)::INT) AS share_of_peers_below
    FROM report_universe AS u
    JOIN latest AS own USING (unitid)
    JOIN peer_pairs AS p USING (unitid)
    JOIN latest AS peer ON peer.unitid = p.peer_unitid
    WHERE u.state_abbr = $state
    GROUP BY ALL
    ORDER BY grad_rate DESC
""", {"state": state}).df()

#| label: overview
OWN, PEERS = "Institution", "Median of its national peers"
order = summary.inst_name.tolist()
y = alt.Y("inst_name:N", title=None, sort=order, axis=alt.Axis(labelLimit=300, labelColor=INK, labelFontSize=12, grid=False, domain=False, ticks=False))
x_scale = alt.Scale(domain=[0, 1])
base = alt.Chart(summary)

link = base.mark_rule(color=PEER, strokeWidth=2, opacity=0.5).encode(
    y=y,
    x=alt.X("grad_rate:Q", title=None, scale=x_scale,
            axis=alt.Axis(format=".0%", orient="top", values=[0, 0.2, 0.4, 0.6, 0.8, 1], gridColor=GRID, domain=False, ticks=False, labelColor=MUTED)),
    x2="peer_median:Q",
)
dots = (
    base.transform_fold(["grad_rate", "peer_median"], as_=["series", "x"])
    .transform_calculate(series=f"datum.series == 'grad_rate' ? {OWN!r} : {PEERS!r}")
    .mark_point(size=120, filled=True, opacity=1, stroke="white", strokeWidth=2)
    .encode(
        y=y, x=alt.X("x:Q", scale=x_scale),
        color=alt.Color("series:N", scale=alt.Scale(domain=[OWN, PEERS], range=[FOCUS, PEER]),
                        legend=alt.Legend(orient="top", title=None, labelColor=INK, labelFontSize=12, labelLimit=300)),
        href="report_file:N",
        tooltip=[
            alt.Tooltip("inst_name:N", title="Institution"),
            alt.Tooltip("carnegie_label:N", title="Peer group"),
            alt.Tooltip("grad_rate:Q", title="Graduation rate", format=".1%"),
            alt.Tooltip("peer_median:Q", title="Peer median", format=".1%"),
            alt.Tooltip("n_peers:Q", title="Peers"),
        ],
    )
)
(link + dots).properties(width=380, height=alt.Step(24)).configure(
    background="transparent", font="system-ui, -apple-system, 'Segoe UI', sans-serif"
).configure_view(stroke=None)
